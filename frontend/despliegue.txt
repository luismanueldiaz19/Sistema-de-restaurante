# 1. Instalar dependencias optimizadas (si no lo has hecho)
composer install --optimize-autoloader --no-dev

# 2. Caché de configuración (muy importante)
php artisan config:cache

# 3. Caché de rutas (acelera la navegación)
php artisan route:cache

# 4. Caché de vistas Blade
php artisan view:cache

# 5. Asegurar permisos de las carpetas de caché y subidas
chown -R www-data:www-data storage bootstrap/cache
chmod -R 775 storage bootstrap/cache

# 6. Crear el link simbólico para las imágenes (si subes logos o fotos de platillos)
php artisan storage:link
